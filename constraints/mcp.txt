# Constraints for the optional `mcp` extra (pyproject.toml: mcp = ["mcp>=2.0,<3.0"]).
#
# `pip install -c constraints/mcp.txt` — a constraints file only pins versions
# for packages that are already part of the install; it does not add them as
# new dependencies on its own.
#
# Why pinned: pyproject.toml's `mcp>=2.0,<3.0` range is intentionally broad (any
# 2.x release), but a floating range means CI resolves whatever 2.x is newest at
# install time. Pinning here keeps the integration job on the version the suite
# is verified against.
#
# mcp 2.2 withholds the text of any exception that is not ToolError (a crash
# becomes "Error executing tool <name>" with no cause). The MCP tools translate
# anticipated ValueError into ToolError so clean messages still reach the
# client; this pin is the version that gate was checked against.
#
# mcp-types is mcp's own runtime dependency (mcp==2.2.0 requires
# mcp-types==2.2.0 exactly), pinned here too for an explicit, self-documenting
# resolution rather than relying on mcp's own metadata.
#
# NOTE: dependabot's pip ecosystem (.github/dependabot.yml) tracks
# pyproject.toml, not this file — a constraints file isn't a dependency
# manifest dependabot understands. When dependabot does bump this file (it
# did for 2.0.0 → 2.2.0), the integration job is the check that the new 2.x
# still surfaces ToolError text.
mcp==2.2.0
mcp-types==2.2.0
